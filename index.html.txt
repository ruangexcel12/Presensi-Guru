    <div class="p-5 space-y-4">
      <div>
        <label class="block text-xs font-semibold text-slate-700 mb-1">Nama Lengkap Guru</label>
        <input type="text" id="namaGuru" placeholder="Masukkan nama Anda" 
          class="w-full px-3 py-2 text-sm border border-slate-300 rounded-xl focus:ring-2 focus:ring-emerald-500 focus:outline-none text-slate-800 transition">
      </div>

      <div>
        <label class="block text-xs font-semibold text-slate-700 mb-1">Kamera Selfie</label>
        <div class="relative bg-slate-900 rounded-xl overflow-hidden aspect-video flex items-center justify-center border border-slate-300">
          <video id="webcam" autoplay playsinline muted class="w-full h-full object-cover"></video>
          <canvas id="canvas" class="hidden"></canvas>
          <div id="cameraStatus" class="absolute bottom-2 left-2 bg-black/60 text-white text-[10px] px-2 py-0.5 rounded-md backdrop-blur-sm">
            Membuka kamera...
          </div>
        </div>
      </div>

      <div class="bg-slate-50 p-3 rounded-xl border border-slate-200 text-xs space-y-1">
        <div class="flex justify-between items-center text-slate-600">
          <span>Status GPS:</span>
          <span id="gpsStatus" class="font-semibold text-amber-600">Mencari Lokasi...</span>
        </div>
        <div class="flex justify-between items-center text-slate-600">
          <span>Koordinat:</span>
          <span id="coordsDisplay" class="font-mono text-slate-500">-</span>
        </div>
      </div>

      <button id="btnPresensi" onclick="prosesPresensi()" disabled
        class="w-full bg-emerald-600 hover:bg-emerald-700 text-white font-semibold py-3 px-4 rounded-xl shadow-lg transition duration-200 disabled:opacity-50 disabled:cursor-not-allowed flex items-center justify-center gap-2 text-sm">
        <i class="fa-solid fa-check-circle"></i>
        <span>Kirim Presensi</span>
      </button>

      <div id="alertBox" class="hidden p-3 rounded-xl text-xs border"></div>
    </div>
  </div>

  <script>
    // MASUKKAN URL DEPLOYMENT APPS SCRIPT (WEB APP) ANDA DI SINI
    const URL_API = "https://script.google.com/macros/s/AKfycbxM2iHQZpmfZBLn7QcgpEM09Xyjkd-_bm2o_fy9EzbiFRQd8_e5U9bKXIn4H0ZbZa4Kxg/exec";

    let userLat = null;
    let userLng = null;
    let isCameraReady = false;

    window.onload = function() {
      aktifkanKamera();
      dapatkanLokasi();
    };

    async function aktifkanKamera() {
      const webcam = document.getElementById('webcam');
      const cameraStatus = document.getElementById('cameraStatus');

      try {
        const stream = await navigator.mediaDevices.getUserMedia({
          video: { facingMode: 'user', width: { ideal: 640 }, height: { ideal: 480 } },
          audio: false
        });
        webcam.srcObject = stream;
        isCameraReady = true;
        cameraStatus.innerText = "Kamera Aktif";
        cameraStatus.className = "absolute bottom-2 left-2 bg-emerald-600/80 text-white text-[10px] px-2 py-0.5 rounded-md backdrop-blur-sm";
        cekSiapKirim();
      } catch (err) {
        cameraStatus.innerText = "Izinkan Akses Kamera di Browser";
        cameraStatus.className = "absolute bottom-2 left-2 bg-red-600/80 text-white text-[10px] px-2 py-0.5 rounded-md backdrop-blur-sm";
      }
    }

    function dapatkanLokasi() {
      const gpsStatus = document.getElementById('gpsStatus');
      const coordsDisplay = document.getElementById('coordsDisplay');

      if (!navigator.geolocation) {
        gpsStatus.innerText = "GPS Tidak Didukung";
        gpsStatus.className = "font-semibold text-red-600";
        return;
      }

      navigator.geolocation.getCurrentPosition(
        (pos) => {
          userLat = pos.coords.latitude;
          userLng = pos.coords.longitude;
          gpsStatus.innerText = "Lokasi Terkunci";
          gpsStatus.className = "font-semibold text-emerald-600";
          coordsDisplay.innerText = `${userLat.toFixed(5)}, ${userLng.toFixed(5)}`;
          cekSiapKirim();
        },
        (err) => {
          gpsStatus.innerText = "Gagal Mengambil GPS";
          gpsStatus.className = "font-semibold text-red-600";
        },
        { enableHighAccuracy: true, timeout: 15000, maximumAge: 0 }
      );
    }

    function cekSiapKirim() {
      const btn = document.getElementById('btnPresensi');
      if (userLat && userLng && isCameraReady) {
        btn.disabled = false;
      } else {
        btn.disabled = true;
      }
    }

    function ambilFoto() {
      const webcam = document.getElementById('webcam');
      const canvas = document.getElementById('canvas');
      canvas.width = webcam.videoWidth || 640;
      canvas.height = webcam.videoHeight || 480;
      const ctx = canvas.getContext('2d');
      ctx.drawImage(webcam, 0, 0, canvas.width, canvas.height);
      return canvas.toDataURL('image/jpeg', 0.8);
    }

    function prosesPresensi() {
      const nama = document.getElementById('namaGuru').value.trim();
      const btn = document.getElementById('btnPresensi');

      if (!nama) {
        tampilkanAlert("Nama guru wajib diisi!", "error");
        return;
      }

      btn.disabled = true;
      btn.innerHTML = `<i class="fa-solid fa-spinner animate-spin"></i> Memproses...`;

      const fotoBase64 = ambilFoto();
      const payload = { namaGuru: nama, lat: userLat, lng: userLng, fotoBase64: fotoBase64 };

      fetch(URL_API, {
        method: 'POST',
        body: JSON.stringify(payload)
      })
      .then(res => res.json())
      .then(res => {
        btn.disabled = false;
        btn.innerHTML = `<i class="fa-solid fa-check-circle"></i> Kirim Presensi`;

        if (res.success) {
          tampilkanAlert(`Presensi Berhasil! Jarak Anda ${res.jarak}m dari sekolah.`, "success");
          document.getElementById('namaGuru').value = '';
        } else {
          tampilkanAlert(`Presensi Ditolak! Anda di luar radius (${res.jarak}m dari sekolah).`, "error");
        }
      })
      .catch(err => {
        btn.disabled = false;
        btn.innerHTML = `<i class="fa-solid fa-check-circle"></i> Kirim Presensi`;
        tampilkanAlert("Gagal terhubung ke server Apps Script.", "error");
      });
    }

    function tampilkanAlert(pesan, tipe) {
      const alertBox = document.getElementById('alertBox');
      alertBox.classList.remove('hidden', 'bg-red-50', 'border-red-200', 'text-red-700', 'bg-emerald-50', 'border-emerald-200', 'text-emerald-700');
      if (tipe === 'error') {
        alertBox.classList.add('bg-red-50', 'border-red-200', 'text-red-700');
      } else {
        alertBox.classList.add('bg-emerald-50', 'border-emerald-200', 'text-emerald-700');
      }
      alertBox.innerText = pesan;
    }
  </script>
