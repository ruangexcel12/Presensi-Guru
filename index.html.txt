      <h1 class="text-2xl font-bold">Presensi Guru</h1>
      <p class="text-xs text-emerald-100 mt-1">Sistem Absensi Berbasis Geolocation</p>
    </div>

    <div class="p-6 space-y-5">
      <div>
        <label class="block text-sm font-semibold text-slate-700 mb-2">Nama Lengkap Guru</label>
        <input type="text" id="namaGuru" placeholder="Masukkan nama Anda" 
          class="w-full px-4 py-3 border border-slate-300 rounded-xl focus:ring-2 focus:ring-emerald-500 focus:outline-none text-slate-800 transition">
      </div>

      <div class="bg-slate-50 p-4 rounded-xl border border-slate-200 text-sm space-y-2">
        <div class="flex justify-between items-center text-slate-600">
          <span>Status GPS:</span>
          <span id="gpsStatus" class="font-semibold text-amber-600">Mencari Lokasi...</span>
        </div>
        <div class="flex justify-between items-center text-slate-600">
          <span>Koordinat Anda:</span>
          <span id="coordsDisplay" class="font-mono text-xs text-slate-500">-</span>
        </div>
      </div>

      <button id="btnPresensi" onclick="kirimPresensi()" disabled
        class="w-full bg-emerald-600 hover:bg-emerald-700 text-white font-semibold py-3.5 px-4 rounded-xl shadow-lg transition duration-200 disabled:opacity-50 disabled:cursor-not-allowed flex items-center justify-center gap-2">
        <i class="fa-solid fa-check-circle"></i>
        <span>Kirim Presensi Sekarang</span>
      </button>

      <div id="alertBox" class="hidden p-4 rounded-xl text-sm border"></div>
    </div>
  </div>

  <script>
    let userLat = null;
    let userLng = null;

    window.onload = function() {
      dapatkanLokasi();
    };

    function dapatkanLokasi() {
      const gpsStatus = document.getElementById('gpsStatus');
      const coordsDisplay = document.getElementById('coordsDisplay');
      const btnPresensi = document.getElementById('btnPresensi');

      if (!navigator.geolocation) {
        gpsStatus.innerText = "GPS tidak didukung";
        gpsStatus.className = "font-semibold text-red-600";
        return;
      }

      navigator.geolocation.getCurrentPosition(
        (pos) => {
          userLat = pos.coords.latitude;
          userLng = pos.coords.longitude;
          gpsStatus.innerText = "Lokasi Terkunci";
          gpsStatus.className = "font-semibold text-emerald-600";
          coordsDisplay.innerText = `${userLat.toFixed(5)}, ${userLng.toFixed(5)}`;
          btnPresensi.disabled = false;
        },
        (err) => {
          gpsStatus.innerText = "Gagal Mengambil Lokasi";
          gpsStatus.className = "font-semibold text-red-600";
          alert("Harap izinkan akses lokasi/GPS di browser Anda.");
        },
        { enableHighAccuracy: true, timeout: 10000, maximumAge: 0 }
      );
    }

    function kirimPresensi() {
      const nama = document.getElementById('namaGuru').value.trim();
      const alertBox = document.getElementById('alertBox');
      const btnPresensi = document.getElementById('btnPresensi');

      if (!nama) {
        tampilkanAlert("Nama guru wajib diisi!", "error");
        return;
      }

      btnPresensi.disabled = true;
      btnPresensi.innerHTML = `<i class="fa-solid fa-spinner animate-spin"></i> Memproses...`;

      google.script.run
        .withSuccessHandler((res) => {
          btnPresensi.disabled = false;
          btnPresensi.innerHTML = `<i class="fa-solid fa-check-circle"></i> Kirim Presensi Sekarang`;

          if (res.success) {
            tampilkanAlert(`Presensi Berhasil! Jarak Anda ${res.jarak}m dari sekolah.`, "success");
            document.getElementById('namaGuru').value = '';
          } else {
            tampilkanAlert(`Presensi Ditolak! Anda berada di luar radius (${res.jarak}m dari sekolah. Maksimum: ${res.maxRadius}m).`, "error");
          }
        })
        .withFailureHandler((err) => {
          btnPresensi.disabled = false;
          btnPresensi.innerHTML = `<i class="fa-solid fa-check-circle"></i> Kirim Presensi Sekarang`;
          tampilkanAlert("Terjadi kesalahan sistem.", "error");
        })
        .catatPresensi(nama, userLat, userLng);
    }

    function tampilkanAlert(pesan, tipe) {
      const alertBox = document.getElementById('alertBox');
      alertBox.classList.remove('hidden', 'bg-red-50', 'border-red-200', 'text-red-700', 'bg-emerald-50', 'border-emerald-200', 'text-emerald-700');
      
      if (tipe === 'error') {
        alertBox.classList.add('bg-red-50', 'border-red-200', 'text-red-700');
      } else {
        alertBox.classList.add('bg-emerald-50', 'border-emerald-200', 'text-emerald-700');
      }
      alertBox.innerText = pesan;
    }
  </script>
